server {
        listen 80;
        server_name inventory-app.natte.site;

        # Redirect all HTTP traffic to HTTPS
        location / {
            return 301 https://$host$request_uri;
        }
    }

    server {
        listen 443 ssl http2;
        server_name inventory-app.natte.site;

        # SSL Certificate paths (ensure these match your Certbot paths if different)
        ssl_certificate /etc/letsencrypt/live/inventory-app.natte.site/fullchain.pem;
        ssl_certificate_key /etc/letsencrypt/live/inventory-app.natte.site/privkey.pem;
        include /etc/letsencrypt/options-ssl-nginx.conf;
        ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

        # Main location for the frontend application
        location / {
            proxy_pass http://inventaris_frontend_app:80;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
        }

        # Location for the backend API calls
        location /api/ {
            # The backend container is exposed on host port 5001.
            proxy_pass http://localhost:5001/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }
    }
